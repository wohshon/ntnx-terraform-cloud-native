read -r -p "Prism Central FQDN or IP: " PC
read -r -p "Prism Central username: " USER
read -r -s -p "Password: " PASS
echo

PC="${PC#https://}"
PC="${PC#http://}"
PC="${PC%/}"
API="https://${PC}:9440/api/iam/v4.0/authn/user-groups"

for GROUP in administrators users; do
    JSON=$(curl -skS -u "$USER:$PASS" --get \
      --data-urlencode "\$filter=name eq '$GROUP'" "$API")

    ID=$(printf '%s' "$JSON" | tr -d '\r\n' |
      sed -n 's/.*"extId"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')

    if [ -z "$ID" ]; then
        echo "Group not found: $GROUP"
        continue
    fi

    HEADER=$(mktemp)

    curl -skS -u "$USER:$PASS" \
      -D "$HEADER" -o /dev/null "$API/$ID"

    ETAG=$(awk 'BEGIN{IGNORECASE=1}
      /^etag:/ {
        sub(/\r$/, "")
        sub(/^[^:]*:[[:space:]]*/, "")
        print
        exit
      }' "$HEADER")

    rm -f "$HEADER"

    if [ -z "$ETAG" ]; then
        echo "Could not retrieve ETag for $GROUP"
        exit 1
    fi

    echo "Found $GROUP — ETag retrieved."
    read -r -p "Delete $GROUP? Type DELETE: " CONFIRM

    if [ "$CONFIRM" = "DELETE" ]; then
        RESULT=$(mktemp)

        STATUS=$(curl -skS -u "$USER:$PASS" -X DELETE \
          -H "If-Match: $ETAG" \
          -H "Ntnx-Request-Id: $(date +%s)-$RANDOM" \
          -w "%{http_code}" \
          -o "$RESULT" \
          "$API/$ID")

        if [[ "$STATUS" == 2* ]]; then
            echo "Removed: $GROUP"
        else
            echo "Failed: $GROUP — HTTP $STATUS"
            cat "$RESULT"
        fi

        rm -f "$RESULT"
    else
        echo "Skipped: $GROUP"
    fi
done
