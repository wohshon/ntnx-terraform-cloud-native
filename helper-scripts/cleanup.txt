# clean up user groups under iam 
(
  # FILL THESE THREE IN:
  PC_IP="YOUR_PRISM_CENTRAL_IP"
  PC_USER="YOUR_USERNAME"
  PC_PASS="YOUR_PASSWORD"

  BASE_URL="https://${PC_IP}:9440/api/nutanix/v3"

  echo "1. Forcing deletion of AD Group: administrators..."
  ADMIN_UUID=$(curl -sk -u "${PC_USER}:${PC_PASS}" -X POST -H "Content-Type: application/json" \
    -d '{"kind":"user_group"}' "$BASE_URL/user_groups/list" \
    | jq -r '.entities[]? | select(.spec.resources.directory_service_user_group.distinguished_name == "cn=administrators,cn=builtin,dc=ntnxlab,dc=local") | .metadata.uuid')
  
  if [ -n "$ADMIN_UUID" ]; then
    curl -sk -u "${PC_USER}:${PC_PASS}" -X DELETE "$BASE_URL/user_groups/$ADMIN_UUID"
    echo " -> Deleted administrators group (UUID: $ADMIN_UUID)."
  else
    echo " -> Not found in API."
  fi

  echo "2. Forcing deletion of AD Group: users..."
  USER_UUID=$(curl -sk -u "${PC_USER}:${PC_PASS}" -X POST -H "Content-Type: application/json" \
    -d '{"kind":"user_group"}' "$BASE_URL/user_groups/list" \
    | jq -r '.entities[]? | select(.spec.resources.directory_service_user_group.distinguished_name == "cn=users,cn=builtin,dc=ntnxlab,dc=local") | .metadata.uuid')
  
  if [ -n "$USER_UUID" ]; then
    curl -sk -u "${PC_USER}:${PC_PASS}" -X DELETE "$BASE_URL/user_groups/$USER_UUID"
    echo " -> Deleted users group (UUID: $USER_UUID)."
  else
    echo " -> Not found in API."
  fi
)
